//////////////////////////////////////////////////////////////////////////////////////-------EVENT
//commande1:
CREATE SEQUENCE evt_seq
START WITH 1
INCREMENT BY 1
NOCACHE
NOCYCLE;

//commande 2:event
CREATE OR REPLACE TRIGGER evt_id_trigger
BEFORE INSERT ON EVT
FOR EACH ROW
BEGIN
    :NEW.ID_EVENEMENT := evt_seq.NEXTVAL;
END;
/
